function startSpeech() {
    const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition;
    if (!SpeechRecognition) {
        alert("您的瀏覽器不支援語音辨識，請使用 Safari 或 Chrome。");
        return;
    }

    const recognition = new SpeechRecognition();
    recognition.lang = 'zh-TW';
    document.getElementById('status').innerText = "狀態：聆聽中...";
    recognition.start();

    recognition.onresult = async (event) => {
        const userText = event.results[0][0].transcript;
        document.getElementById('transcript').innerText = `「${userText}」`;
        document.getElementById('status').innerText = "狀態：印象派繪製中 (約需 3~5 秒)...";

        try {
            // 呼叫我們自訂的 Vercel 後端，完全不洩漏 Key
            const response = await fetch('/api/generate', {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({ text: userText })
            });

            const data = await response.json();
            if (data.imageUrl) {
                document.getElementById('story-img').src = data.imageUrl;
                document.getElementById('status').innerText = "狀態：繪本生成成功！";
            } else {
                throw new Error(data.error);
            }
        } catch (err) {
            console.error(err);
            document.getElementById('status').innerText = "狀態：生成失敗，請再試一次。";
        }
    };
}
</script>
